# Debian 13 (trixie) runtime package manifest for the Retro PC appliance.
#
# This list is intentionally limited to packages needed by the physical
# appliance at runtime and for normal maintenance. It is suitable as the
# explicit package list layered on top of a minimal Debian 13 installation.
# Dependency packages pulled in by APT are not listed separately.
#
# The official debian:13-slim image is a useful reference for keeping the
# starting point small, but the final target is a physical Debian system with
# systemd, hardware devices, SSH, and audio support.

# Base service/runtime
systemd
systemd-sysv
udev
dbus
e2fsprogs
# grub-efi-amd64 installs grub-install with EFI support for the GPT/UEFI target disk.
grub-efi-amd64
efibootmgr
systemd-zram-generator
plymouth
plymouth-themes
systemd-resolved

# Appliance administration and file transfer
openssh-server
sudo

# Hardware and audio runtime
alsa-utils
kmod
fuse3
libgles2
libgbm1
libdrm2
libdrm-tests
libegl-mesa0
mesa-libgallium
libgl1-mesa-dri
libegl1

# Small maintenance/networking toolset
procps
iproute2
util-linux
pciutils
mtools

# WiFi (first-boot configured via wpa_supplicant + systemd-networkd + dialog).
# firmware-realtek ships the rtw88/rtw88xx blobs for the TPLink 2.4GHz dongle
# under the non-free-firmware component, which the installer enables via
# mmdebstrap --components=main,non-free-firmware. wpasupplicant is the
# association backend because systemd-networkd does not support WPA2-PSK
# natively (its [WiFi] section + PreSharedKey= would be silently ignored);
# `iw` is the diagnostic tool.
dialog
firmware-realtek
wpasupplicant
iw

ca-certificates

# QEMU host stack for the Windows XP VM (ADR 0008). QEMU's GTK display
# (gl=on) runs in a bare Xorg; SDL on KMS/DRM conflicted with dxvk-native.
# Installed without Recommends: the root partition is kept small (3 GiB).
# xserver-xorg-legacy ships the setuid Xorg.wrap, because the appliance has no
# logind sessions to hand Xorg the DRM and input devices. The modesetting
# video driver is built into xserver-xorg-core.
xserver-xorg-core
xserver-xorg-input-libinput
xserver-xorg-legacy
xinit
# Kiosk window manager that honours QEMU's fullscreen request.
matchbox-window-manager
# GTK 3 runtime for QEMU's display. dbus-x11 gives GTK's dconf a session bus
# without dbus-user-session, which would pull in libpam-systemd and start
# logind sessions, changing console/Xorg behaviour for retrobox-boot and 86Box.
libgtk-3-0t64
dbus-x11
# Vulkan for dxvk-native (SVGA3D -> Vulkan). The build trims
# mesa-vulkan-drivers to Intel's ICD with dpkg-vulkan-intel-only.cfg.
libvulkan1
mesa-vulkan-drivers
# User-mode NAT networking (-netdev user) for the XP guest.
libslirp0

# Deliberately excluded from this base manifest:
# - .NET SDK, compilers, and build tools
# - Docker or container tooling
# - desktop environments, display managers, and full window managers (the
#   bare Xorg + matchbox above exists only to host QEMU's fullscreen window)
# - SDL and Wayland packages
# - DOS, Windows, drivers, games, or other copyrighted media
#
# 86Box is supplied as an AppImage and QEMU as a tarball under /opt (both
# pinned in ../86box.env). Host libraries they need beyond this list must be
# added here explicitly, with the reason, rather than speculatively.
