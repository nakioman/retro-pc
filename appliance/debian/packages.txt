# Debian 13 (trixie) runtime package manifest for the Retro PC appliance.
#
# This list is intentionally limited to packages needed by the physical
# appliance at runtime and for normal maintenance. It is suitable as the
# explicit package list layered on top of a minimal Debian 13 installation.
# Dependency packages pulled in by APT are not listed separately.
#
# The official debian:13-slim image is a useful reference for keeping the
# starting point small, but the final target is a physical Debian system with
# systemd, hardware devices, SSH, Samba, and audio support.

# Base service/runtime
systemd
# systemd-sysv provides /sbin/init (and reboot/halt/poweroff). The minbase build
# variant is essential-only and does not pull it in, so the target must list it
# explicitly or run-init fails with "/sbin/init: no such file or directory".
systemd-sysv
udev

# Filesystem tooling. e2fsprogs is Priority: required but not Essential, so
# minbase omits it — without it the initramfs has no fsck.ext4 and boot fails
# checking the ext4 root.
e2fsprogs

# System message bus. systemd-logind needs it to handle the ACPI power button
# (HandlePowerKey=poweroff) and manage login sessions; minbase omits it, which
# leaves the power button dead.
dbus

# Bootloader for the installed target disk (BIOS/legacy MBR).
# grub-install runs from the live installer; grub-pc keeps update-grub working
# on the deployed system.
grub-pc

# Quiet appliance boot splash (a minimal loading bar). Linux stays an
# implementation detail, so first paint should not be a wall of boot text.
plymouth
plymouth-themes

# Appliance administration and file transfer
openssh-server
samba
sudo

# Hardware and audio runtime
alsa-utils
kmod
fuse3

# Small maintenance/networking toolset
procps
iproute2
util-linux
pciutils
ca-certificates

# Deliberately excluded from this base manifest:
# - .NET SDK, compilers, and build tools
# - Docker or container tooling
# - desktop environments and window managers
# - X11/Wayland/SDL/DRM/graphics packages for 86Box
# - DOS, Windows, drivers, games, or other copyrighted media
#
# 86Box is supplied as an AppImage. Its graphics dependencies and any
# additional host libraries required by the selected AppImage must be decided
# during the physical graphics integration work, not added speculatively here.
